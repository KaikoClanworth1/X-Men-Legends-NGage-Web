## 5. Pseudocode
```js
OnStart() { api.FadeIn(this, onFadedIn); }              // 0x10057e5c
onFadedIn() {                                           // 0x10057e84
  if (this.fc8) return;
  this.fc8 = 1;
  api.SetCharTypeLevel("bhbrawler", 3, 1);
  api.SetCharTypeLevel("bhpsionic", 3, 1);
  ["p1","p2","p3","p4","p5"].forEach(n => api.SetFlag400(n, 1));
}

onKnockBillboard(evt) {                                 // 0x10057f40
  if (evt.code != 0 || this.fd0) return;
  this.fd0 = 1;
  api.MakePartyTarget("character3"); api.MakePartyTarget("character4");
}
onKnockAnt(evt) {                                       // 0x10057f94
  if (evt.code != 0 || this.fd4) return;
  this.fd4 = 1;
  api.MakePartyTarget("character5"); api.MakePartyTarget("character6");
}

onEndLevel(evt) {                                       // 0x10057fe8 (no once-guard)
  if (evt.code != 0) return;
  api.SetObjective(1, 2, 0);          // "…" completed, no popup
  api.StartDialogue("Cyclops-2", this, afterCyclops2, 0);
  // Cyclops: let's head back home. END result 0 (not used)
}
afterCyclops2(res) { api.FadeOut(this, afterFade); }    // 0x1005805c
afterFade() {                                           // 0x10058084
  for (let s = 0; s < 4; s++) api.SetHealthId(this.GetPartyMemberId(s), 1500);
  api.ChangeLevel("e01cXJ", 1);       // X-Jet cutscene map
}

onToStreets(evt) {                                      // 0x1005811c
  switch (evt.code) {
    case 0: api.SetActionPromptId(evt.subject.GetId(), 1); api.ShowMessage(1, 1); break; // "…"
    case 3: api.SetActionPromptId(evt.subject.GetId(), 0); api.ShowMessage(1, 0); break;
    case 2: api.FadeOut(this, () => api.ChangeLevel("e01m01", 2)); break;             // cb 0x100581d8
  }
}

onItem(evt) {                                           // 0x100581ec
  if (this.fcc || evt.code != 7) return;
  this.fcc = 1;
  api.SetDialogueSpeakerEntity(evt.subject);
  api.StartDialogue("getbomb", this, null, 1);  // (Player): this can help in a tough battle. result 0
}
```

## 6. Flow summary
- Arrive at spawn1 from e01m01. First fade-in: brawlers/psionics to level 3, p1..p5 SetFlag400.
- No AddObjective here; objective 1 (added in e01m01) completes at any endLevel zone.
- Billboard/antenna zones make the party engage ambushers.
- Win: endLevel* -> Cyclops-2 -> FadeOut -> party HP 1500 -> `ChangeLevel("e01cXJ", 1)`.
- Back: toStreets* action -> `ChangeLevel("e01m01", 2)`.
- Heroes unlocked (Init): Storm, Rogue, Wolverine, Cyclops, Iceman.
