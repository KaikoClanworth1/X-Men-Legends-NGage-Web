## 5. Pseudocode

```js
OnStart() {                                   // 0x10077ea8
  if (this.fc8 == 0) {
    this.fc8 = 1;
    SetFacing("colossus", 3);
    SetNeutral("colossus"); SetNeutral("Technician"); SetNeutral("Tech2");
    MapSet1(94, 41, 1);                        // lock trigger5 door tile
    introStep();
  } else FadeIn(this, null);
  if (GetVarC0() == 3) this.fe8 = 1;           // valves turned on in e09m17
}

advance() {                                    // 0x100781f0 (0x10078228 = thunk)
  if (this.fcc == 0) { this.fd0++; introStep(); }
  else if (this.fcc == 1) { this.fd4++; endStep(); }
}

introStep() { switch (this.fd0) {              // 0x10077f54
  case 0: this.fcc=0; SetCinematicMode(0,0); SetHudVisible(0); SetPlayerControl(0);
          FadeIn(this, advance); break;
  case 1: this.fcc=0; SetDialogueSpeakerName(GetPartyMemberId(0));
          StartDialogue("BeginMission", this, advance, 0); break;
          // Player: Brotherhood attacked a nuclear facility in broad daylight?
  case 2: this.fcc=0; WalkToTile("Technician", 127, 31, 0); break;   // → event 5 → advance
  case 3: this.fcc=0; for (s=0..3) FaceEntityIdName(GetPartyMemberId(s), "Technician");
          CameraFollowEntity("Technician", 0x2ff);
          StartDialogue("TechConv", this, advance, 0); break;
          // Technician begs for life; "…" (Colossus) is at the nuclear core. END 0
  case 4: this.fcc=0; CameraFollowPlayer(0x2ff); SetCinematicMode(1,0); SetHudVisible(1);
          SetPlayerControl(1);
          WalkToTile("Technician", 133, 31, -1);
          AddObjective(18 /*Find Colossus*/, 1); break;
}}

Technician(evt) {                               // 0x100781a8
  if (evt.code != 5) return;
  if (this.fd0 == 4) RemoveEntity("Technician");
  this.fcc = 0; advance();                        // fd0 2→3, or 4→5 (no case: no-op)
}

colossus(evt) {                                 // 0x100788e0
  if (evt.code == 0 && this.ff0 == 0) endStep(); // fd4 == 0
}

endStep() { switch (this.fd4) {                 // 0x100788fc
  case 0: SetObjective(18, 2, 1);
          StartDialogue("MeetCollossus", this, null, 0);
          // Colossus holding the regulator; reactor at critical mass. END 0
          Teleport("tech2", 1000, 5750, 0);      // lab tech moved near mainvalve
          this.ff0 = 1; break;                  // waits for mainvalve
  case 1: this.fcc=1; SetPlayerControl(0); FadeOut(this, advance); break;
  case 2: this.fcc=1; this.fdc=0;               // stop smoke timer
          SetHudVisible(0); RemoveEntity("tech2"); RemoveEntity("Colossus");
          SpawnEntity("Colossus2", "colossus03", 0, 1900, 1050, 0, 0);
          TeleportId(GetPartyMemberId(0), 2080, 1080, 4);
          TeleportId(GetPartyMemberId(1), 2080, 1180, 3);
          TeleportId(GetPartyMemberId(2), 2080, 1280, 3);
          TeleportId(GetPartyMemberId(3), 2080,  880, 5);
          CameraPanTo(2000, 1100, 0x1000);
          FadeIn(this, advance); break;
  case 3: this.fcc=1; SetDialogueSpeakerName(GetPartyMemberId(0));
          StartDialogue("EndConv", this, advance, 0); break;
          // Colossus: Magneto + Acolytes making gravitic-ion material in the reactor;
          // Illyana is ill → take her to Muir Island. END 0
  case 4: this.fcc=1; FadeOut(this, advance); break;
  case 5: this.fcc=1; RemoveEntity("Colossus2");
          SpawnEntity("colossus", "colossus", 0, 1900, 1050, 0, 0);
          SetCharTypeLevel("colossus", GetEntityLevel(GetEntity(GetPartyMemberId(0))), 1);
          SetAlly("colossus");
          CompleteEpisode(1); break;
}}

Tech2(evt) {                                     // 0x1007822c
  // gate: IsPartyNotEmpty() must return 0 (see Unknowns)
  switch (evt.code) {
  case 0: if (!IsPartyNotEmpty() && this.ff0 == 1) SetActionPromptId(evt.subject.id, 1); break;
  case 3: if (!IsPartyNotEmpty() && this.ff0 == 1) SetActionPromptId(evt.subject.id, 0); break;
  case 2:
    if (IsPartyNotEmpty() || this.ff0 != 1) return;
    FaceEntityId(evt.subject.id, evt.other.id); FaceEntityId(evt.other.id, evt.subject.id);
    if (this.fe8 != 0) return;
    SetDialogueSpeakerEntity(evt.subject);
    StartDialogue("meettech", this, null, 0);
    // Lab tech: reactor overheating; open all coolant valves in adjacent area, then
    // come back and turn on the main coolant valve. END 0
    if (this.fec == 0) { this.fec = 1; AddObjective(63 /*Turn on coolant valves.*/, 1); }
  }
}

goto17(evt) {                                    // 0x10078c88  (goto17a..d)
  switch (evt.code) {
  case 0: if (this.fec == 1) { SetActionPromptId(evt.subject.id, 1); ShowMessage(58 /*To Downstairs*/, 1); }
          else ShowMessage(2 /*Not Ready!*/, 1); break;
  case 3: if (this.fec == 1) { SetActionPromptId(evt.subject.id, 0); ShowMessage(58, 0); }
          else ShowMessage(2, 0); break;
  case 2: if (this.fec == 1) FadeOut(this, () => ChangeLevel("e09m17", 1));  // cb 0x10078d80
  }
}

mainvalve(evt) {                                 // 0x10078384
  switch (evt.code) {
  case 0: SetActionPromptId(evt.subject.id, 1); ShowMessage(156 /*Turn on main coolant valve.*/, 1); break;
  case 3: SetActionPromptId(evt.subject.id, 0); ShowMessage(156, 0); break;
  case 2:
    if (this.fe8 == 1) {
      RemoveEntityId(evt.other.id);              // the valve zone
      ShowPopupText(157 /*Main coolant valve opened.*/);
      RemoveEntity("trigger1");
      WalkTo("Tech2", 1150, 5350, 6);
      SetObjective(66 /*Turn on main coolant valve.*/, 2, 1);
      SetCinematicMode(0, 0);
      this.fcc = 1; advance();                   // fd4 0→1
    } else ShowPopupText(2 /*Not Ready!*/);
  }
}

trigger5(evt) {                                  // 0x10078d94
  if (evt.subject.typeKey == "gambit") {
    switch (evt.code) {
    case 0: SetActionPromptId(evt.subject.id, 1); ShowMessage(55 /*Pick Lock*/, 1); break;
    case 3: SetActionPromptId(evt.subject.id, 0); ShowMessage(55, 0); break;
    case 2: MapSet1(94, 41, 0); ClearMessages(); ShowPopupText(53 /*Door Unlocked!*/);
            RemoveEntity("trigger5"); GiveQuestXP(1000); ShowPopupText(122 /*1000 Quest XP*/);
    }
  } else if (!IsInParty("gambit")) {
    if (evt.code == 0) ShowMessage(52 /*Locked*/, 1);
    else if (evt.code == 3) ShowMessage(52, 0);
  } else if (evt.code == 0) {
    StartDialogue("GambitStartsUnlock", this, null, 0); // Gambit: "…"
  }
}

container(evt) {                                 // 0x10078f1c (container2,3,5,6,7,8)
  switch (evt.code) {
  case 0: SetActionPromptId(evt.subject.id, 1); ShowMessage(158 /*Search*/, 1); break;
  case 3: SetActionPromptId(evt.subject.id, 0); ShowMessage(158, 0); break;
  case 2:
    ClearMessages();
    let item = null;
    if (this.fe0 <= 6) {
      const r = rand16() % 100;
      if (r > 96) item = "…";
      else if (r > 91) item = UNKNOWN_A;          // literal @0x10079040 unresolved
      else if (r > 81) item = UNKNOWN_B;          // literal @0x10079058 unresolved
      else if (r > 66) item = "hypnobomb";
    }
    if (item) { GiveItem(item); this.fe0++; } else ShowPopupText(159 /*Empty*/);
    RemoveEntity(evt.other.name);
  }
}

container1(evt) {                                // 0x100790ac
  // 0/3: SetActionPromptId(subject, 1/0); ShowMessage(158 /*Search*/, 1/0)
  // 2:   ClearMessages(); GiveItem("…"); RemoveEntityId(evt.other.id);
}

smokeTimer() {                                   // timer 0, 0x100784c0
  if (this.fdc == 0) return;                     // not restarted
  const r = rand16() % 13;  this.fdc = r + 1;
  // SpawnEmitter(name, "spark2", x, y, 0, dur)
  [ ["smoke0",1900,1100,3], ["smoke1",2000,1200,3], ["smoke2",2100,1300,3], ["smoke3",2200,1400,3],
    ["smoke4",2250,1500,3], ["smoke5",2250,1500,4], ["smoke6",2150,1400,4], ["smoke7",2050,1300,4],
    [UNKNOWN /*prob. "smoke8"*/,1950,1200,4], ["smoke9",2250,1400,2], ["smoke10",2150,1300,2],
    ["smoke11",2050,1200,2], ["smoke12",1950,1100,2] ][r];
  StartTimer(0, 500 + rand16() % 1000, 0);
}
// rand16(): seed = (seed*15737 + 15149) & 0xFFFF at GetGlobalGame()+0x10
```

## 6. Flow summary
1. First entry: fade in → "BeginMission" → Technician walks to tile (127,31) → "TechConv" → control returned, Technician walks to (133,31) and is removed → **objective 18 "…"**.
2. Optional: container1 → Gem of the Clouds; Gambit picks lock at trigger5 (tile 94,41 cleared, 1000 Quest XP).
3. Reach Colossus (proximity) → objective 18 complete, "MeetCollossus"; `tech2` teleported to (1000,5750); ff0=1.
4. Talk to Tech2 → "meettech", **objective 63 "…"**; fec=1 unlocks `goto17a..d` → `ChangeLevel("e09m17", 1)`.
5. Valves done in e09m17; on re-entry OnStart sees `GetVarC0()==3` → fe8=1 (objective 66 is not added in this script — expected in e09m17).
6. **Win**: mainvalve action with fe8==1 → popup 157, objective 66 complete → fade → freed-Colossus scene ("EndConv") → Colossus spawned as ally at leader's level → **`CompleteEpisode(1)`** (→ idx 10 e10m12, spawn1). No `SetNextMovie` / `UnlockHero` in this script.
