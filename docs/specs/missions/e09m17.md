## Pseudocode
```js
OnStart() {                       // 0x10079864
  FadeIn(this, null);
  StartTimer(0, 60000, 1);        // repeats every 60 s
}

onGoBack(evt) {                   // 0x100798a4
  if (evt.code == 0) { SetActionPromptId(evt.subject.id, 1); ShowMessage(59, 1); } // "…"
  else if (evt.code == 3) { SetActionPromptId(evt.subject.id, 0); ShowMessage(59, 0); }
  else if (evt.code == 2) FadeOut(this, goBackFaded);
}
goBackFaded() {                   // 0x10079960
  if (this.fd4 <= 3) ChangeLevel("e09m16", 2);
  else               ChangeLevel("e09m16", 3);   // all 4 valves open
}

onForcefield(evt) {               // 0x10079994
  if (evt.code == 0) { SetActionPromptId(evt.subject.id, 1); ShowMessage(50, 1); } // "…"
  else if (evt.code == 3) { SetActionPromptId(evt.subject.id, 0); ShowMessage(50, 0); }
  else if (evt.code == 2) {
    ShowPopupText(51);            // "…"
    RemoveEntity("forcefield1"); RemoveEntity("forcefield2");
    MapSet4(11, 41); MapSet4(12, 41); MapSet4(13, 41);   // 0x1002ab34
    MapSet0(11, 41, 0); MapSet0(12, 41, 0); MapSet0(13, 41, 0); // 0x1002a9c0, see Unknowns
  }
}

onValve(evt) {                    // 0x10079adc
  if (evt.code == 0) { SetActionPromptId(evt.subject.id, 1); ShowMessage(154, 1); } // "…"
  else if (evt.code == 3) {
    SetActionPromptId(evt.subject.id, 0); ShowMessage(154, 0);
    if (this.fd4 == 4) {
      SetObjective(63, 2, 1);     // "…" completed
      AddObjective(66, 1);        // "…"
    }
  } else if (evt.code == 2) {
    this.fd4++;
    ShowPopupText(155);           // "…"
    RemoveEntityId(evt.other.id); // evt+4
  }
}

function rand16() {               // inline LCG on GetGlobalGame()+0x10
  game.seed = (game.seed * 15737 + 15149) & 0xFFFF; return game.seed;
}

onContainer(evt) {                // 0x10079bcc
  if (evt.code == 0) { SetActionPromptId(evt.subject.id, 1); ShowMessage(158, 1); } // "Search"
  else if (evt.code == 3) { SetActionPromptId(evt.subject.id, 0); ShowMessage(158, 0); }
  else if (evt.code == 2) {
    ClearMessages();
    let item = null;
    if (this.fd0 <= 6) {
      const r = rand16() % 100;
      if (r > 96) item = "…";
      else if (r > 91) item = "…";
      else if (r > 81) item = "…";
      else if (r > 66) item = "hypnobomb";
    }
    if (item) { GiveItem(item); this.fd0++; }
    else ShowPopupText(159);      // "Empty"
    RemoveEntityId(evt.other.id);
  }
}

onPowerSupplyContainer(evt) {     // 0x10079d5c (container5)
  if (evt.code == 0) SetActionPromptId(evt.subject.id, 1);     // no HUD message
  else if (evt.code == 3) SetActionPromptId(evt.subject.id, 0);
  else if (evt.code == 2) { GiveItem("…"); RemoveEntityId(evt.other.id); }
}

onSpawnTimer() {                  // 0x10079df8, timer 0, every 60 s
  const XS = [1300, 2000, 800, 4200, 1500];  // 0x100fc15c
  const YS = [4300, 2900, 2500, 3700, 2100]; // 0x100fc170
  if (this.fcc > 20) { StopTimer(0); return; }
  const [px, py] = GetPlayerEntity().GetPos();
  this.fc8 = rand16() % 5;
  const x = XS[this.fc8], y = YS[this.fc8];
  // skip if the leader is strictly within ±1000 of the point on both axes
  if (px > x - 1000 && px < x + 1000 && py > y - 1000 && py < y + 1000) return;
  const id = SpawnEntity("testChar" + this.fcc++, "bhbrawler", 0, x, y, 0, 0);
  SetCharTypeLevel("bhbrawler", 13, 1);
  SetFlag400Id(id, 1);
}
```
Item literals were resolved from the binary's literal pool (0x10079cd8, 0x10079cf0, 0x10079d08, 0x10079d30), because pseudo shows only "hypnobomb".

## Flow summary
1. Arrive from e09m16 at spawn1 (2014,5186). Objective 63 "…" should already be active; it is never added here.
2. Every 60 s the timer spawns an extra bhbrawler (`testChar0..20`, up to 21 in total) at one of 5 points, but only if the leader is not near that point. Spawned enemies have no handlers.
3. The forcefield1/2 action opens the forcefield tiles (11..13, 41).
4. Open valve1..4 with the action button (each zone is removed). When the party leaves a valve zone with 4 valves open, objective 63 completes and objective 66 "…" is added.
5. container5 gives "…", presumably for e09m16. The other containers give random consumables, at most 7 items in total.
6. GoBack1..4 action → FadeOut → ChangeLevel("e09m16", spawn 2 if fewer than 4 valves are open, else spawn 3).
- This script has no win condition, CompleteEpisode, movie or hero unlock.
