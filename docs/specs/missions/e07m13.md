## Pseudocode
```js
OnStart() { FadeIn(this, introFade); }                    // 0x1006e858

introFade() {                                              // 0x1006e880
  if (this.fd0) return;
  this.fd0 = 1;
  SetHudVisible(0);
  SetDialogueSpeakerName(GetPartyMemberId(0));  // receives a party id (see Unknowns)
  StartDialogue("init", this, dlgDone, 0);      // Player: stop Brotherhood, protect Arbiter crew (END 0)
}

dlgDone(result) {                                          // 0x1006ebf4
  SetHudVisible(1);
  switch (result) {
    case 0:    if (!this.fc8) { this.fc8 = 1; AddObjective(14, 1); } break; // "…"
    case 1000: WalkToTile("blob", 23, 0, -1);  // fallthrough
    case 900:  CameraFollowPlayer(0x2ff); break;
  }
}

sentZone(evt) {                                            // 0x1006e90c
  if (evt.code != 0) return;
  RemoveEntity("sent1"); RemoveEntity("sent2"); RemoveEntity("sent3"); RemoveEntity("sent4");
  MakePartyTarget("senscout1"); MakePartyTarget("senscout2");
  MakePartyTarget("character900"); MakePartyTarget("character901");
}

senscout1(evt) {                                           // 0x1006e9ac
  if (evt.code != 4) return;
  SetDialogueSpeakerEntity(GetPlayerEntity());
  StartDialogue("toad", this, null, 0);  // Player: Toad was right, Sentinels rebuilt, why attack us and not the Brotherhood? (END 900, no cb)
}

blob(evt) {                                                // 0x1006ea08
  if (!this.fcc && evt.code == 0) {                        // any meetblob zone entered
    this.fcc = 1;
    SetHudVisible(0);
    SetObjective(14, 2, 1);                                // Stop the Brotherhood's attack: done
    AddObjective(15, 1);                                   // "…"
    CameraPanTo(pos("blob").x, pos("blob").y, 0x2ff);
    StartDialogue("meetblob", this, dlgDone, 0);           // Blob taunts, "…" (END 1000)
  }
  if (GetName(evt.subject) == "blob" && evt.code == 5)     // walk to tile (23,0) finished
    Teleport("blob", 2050, 150, 0);                        // out of play
}

revelation(evt) {                                          // 0x1006eb38
  if (evt.code != 0) return;
  SetHudVisible(0);
  SetDialogueSpeakerEntity(evt.subject);
  StartDialogue("muir", this, dlgDone, 0);  // Player: Brotherhood raided Muir for Forge's masking device to hide from Sentinels (END 0)
  RemoveEntity("revelation1".."revelation4");
}

nextlevel(evt) {                                           // 0x1006ec98
  switch (evt.code) {
    case 0: SetActionPromptId(evt.subject.id, 1); ShowMessage(63, 1); break;  // "…"
    case 3: SetActionPromptId(evt.subject.id, 0); ShowMessage(63, 0); break;
    case 2: FadeOut(this, () => ChangeLevel("e07m14", 1)); break;            // cb 0x1006ed54
  }
}
```

## Flow
1. Fade in → "init" → objective 14 "…".
2. Crossing sent1..4 → party auto-targets senscout1/2, character900/901; killing senscout1 → "toad" line.
3. meetblob zones → objective 14 complete, add 15 "…"; Blob dialogue, Blob walks to tile (23,0) and is teleported to (2050,150).
4. revelation zones (near stairs) → "muir" monologue.
5. nextlevel1..4 (action) → ChangeLevel("e07m14", spawn 1). No heroes unlocked, no episode completion.
